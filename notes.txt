Status:

everything can be: unknown
a component can be: unreachable, not-installed, active, inactive


state transitions:
  unknown       -> (state was never checked, software is unreachable)       -> (state is checked) -> unreachable
  unknown       -> (state was never checked, software is installed)         -> (state is checked) -> not-installed
  unknown       -> (state was never checked, software is not running)       -> (state is checked) -> inactive
  unknown       -> (state was never checked, software is running)           -> (state is checked) -> active

  unreachable   -> (previous check said unreachable, this one too)          -> (state is checked) -> unreachable
  unreachable   -> (previous check said unreachable, but now not-installed) -> (state is checked) -> not-installed
  unreachable   -> (previous check said unreachable, but now inactive)      -> (state is checked) -> inactive
  unreachable   -> (previous check said unreachable, but now active)        -> (state is checked) -> active

  not-installed -> (user installs software, but now were unreachable)       -> (state is checked) -> unreachable
  not-installed -> (user installs software, something went wrong)           -> (state is checked) -> not-installed
  not-installed -> (user installs software, which is not auto-started)      -> (state is checked) -> inactive
  not-installed -> (user installs software, which is auto-started)          -> (state is checked) -> active

  inactive      -> (state is checked, but now were unreachable)             -> (state is checked) -> unreachable
  inactive      -> (user uninstalls the inactive software)                  -> (state is checked) -> not-installed
  inactive      -> (user starts software, and fails)                        -> (state is checked) -> inactive
  inactive      -> (user starts software, and is succesful)                 -> (state is checked) -> active

  active        -> (state is checked, but now were unreachable)             -> (state is checked) -> unreachable
  active        -> (user uninstalls the active software)                    -> (state is checked) -> not-installed
  active        -> (user stops software, and is succesful)                  -> (state is checked) -> inactive
  active        -> (user stops software, and fails)                         -> (state is checked) -> active
